     1→import { escapeHtml, hexToRgbStr, type MenuData } from '@/lib/menu-utils';
     2→
     3→/**
     4→ * Construye el HTML completo autocontenido del menú público.
     5→ * Usa concatenación de strings (no template literals anidados) para evitar
     6→ * problemas de escape. Inyecta los datos via JSON.stringify().
     7→ *
     8→ * Soporta tema personalizables (plan Pro): layout, image_size, card_style,
     9→ * font, secondary color, dark mode, cover image, search, etc.
    10→ */
    11→export function buildMenuHTML(data: MenuData): string {
    12→  // Defaults seguros (plan Free = single col, medium img, expanded card, dark)
    13→  const layout = data.theme_layout || 'single';
    14→  const imageSize = data.theme_image_size || 'medium';
    15→  const cardStyle = data.theme_card_style || 'expanded';
    16→  const font = data.theme_font || 'Inter';
    17→  const darkMode = data.theme_dark_mode !== false; // default true
    18→  const showSearch = data.theme_show_search !== false;
    19→  const showCatIcons = data.theme_show_category_icons !== false;
    20→  const rounded = data.theme_rounded_corners !== false;
    21→  const coverUrl = data.theme_cover_url || null;
    22→  const secondary = data.theme_color_secondary || '#1a1a2e';
    23→
    24→  const css = buildCSS({ layout, imageSize, cardStyle, font, darkMode, showSearch, showCatIcons, rounded, coverUrl, secondary });
    25→  const js = buildJS({ layout, imageSize, cardStyle, showSearch });
    26→  const colorRgb = hexToRgbStr(data.color);
    27→  const secondaryRgb = hexToRgbStr(secondary);
    28→
    29→  // Font CDN (solo si no es Inter/system)
    30→  const fontImport = ['Inter', 'system-ui', '-apple-system'].includes(font)
    31→    ? ''
    32→    : `<link rel="preconnect" href="https://fonts.googleapis.com"><link rel="preconnect" href="https://fonts.gstatic.com" crossorigin><link href="https://fonts.googleapis.com/css2?family=${encodeURIComponent(font)}:wght@400;500;600;700;800&display=swap" rel="stylesheet">`;
    33→
    34→  let html = '';
    35→  html += '<!DOCTYPE html>\n';
    36→  html += '<html lang="es">\n';
    37→  html += '<head>\n';
    38→  html += '<meta charset="UTF-8">\n';
    39→  html += '<meta name="viewport" content="width=device-width, initial-scale=1.0, viewport-fit=cover">\n';
    40→  html += '<meta name="theme-color" content="' + data.color + '">\n';
    41→  html += '<meta name="description" content="' + escapeHtml(data.description || data.name) + '">\n';
    42→  html += '<title>' + escapeHtml(data.name) + '</title>\n';
    43→  html += fontImport + '\n';
    44→  html += '<style>' + css + '</style>\n';
    45→  html += '</head>\n';
    46→  html += '<body>\n';
    47→  html += '<div id="app"></div>\n';
    48→  html += '<script>\n';
    49→  html += 'var RESTAURANT = ' + JSON.stringify(data) + ';\n';
    50→  html += 'var SHOW_BRANDING = ' + (!!data.branding_text) + ';\n';
    51→  html += 'var BRANDING_TEXT = ' + JSON.stringify(data.branding_text || '') + ';\n';
    52→  html += 'var THEME = ' + JSON.stringify({ layout, imageSize, cardStyle, showSearch }) + ';\n';
    53→  html += 'document.documentElement.style.setProperty("--accent", "' + data.color + '");\n';
    54→  html += 'document.documentElement.style.setProperty("--accent-rgb", "' + colorRgb + '");\n';
    55→  html += 'document.documentElement.style.setProperty("--secondary", "' + secondary + '");\n';
    56→  html += 'document.documentElement.style.setProperty("--secondary-rgb", "' + secondaryRgb + '");\n';
    57→  html += 'document.documentElement.style.setProperty("--font-main", "' + font + '", "Inter", sans-serif);\n';
    58→  html += js + '\n';
    59→  html += '</scr' + 'ipt>\n';
    60→  html += '</body>\n';
    61→  html += '</html>';
    62→  return html;
    63→}
