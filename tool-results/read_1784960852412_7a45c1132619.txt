    74→  async function handleCreate() {
    75→    if (!newName.trim()) {
    76→      toast.error('Ingresa un nombre');
    77→      return;
    78→    }
    79→    setCreating(true);
    80→    try {
    81→      const res = await fetch('/api/menus', {
    82→        method: 'POST',
    83→        headers: { 'Content-Type': 'application/json' },
    84→        body: JSON.stringify({ name: newName }),
    85→      });
    86→      const data = await res.json();
    87→      if (!res.ok) throw new Error(data.error || 'Error');
    88→      toast.success('Menú creado');
    89→      router.push(`/dashboard/${data.menu.id}`);
    90→    } catch (err) {
    91→      toast.error(err instanceof Error ? err.message : 'Error');
    92→    } finally {
    93→      setCreating(false);
    94→    }
    95→  }

    97→  async function handleDelete(id: string, name: string) {
    98→    if (!confirm(`¿Eliminar el menú "${name}"? Esta acción no se puede deshacer.`)) return;
    99→    try {
   100→      const res = await fetch(`/api/menus/${id}`, { method: 'DELETE' });
   101→      if (!res.ok) throw new Error('Error');
   102→      toast.success('Menú eliminado');
   103→      router.refresh();
   104→    } catch (err) {
   105→      toast.error(err instanceof Error ? err.message : 'Error');
   106→    }
   107→  }

   109→  async function handleLogout() {
   110→    await fetch('/api/auth/logout', { method: 'POST' });

   112→    router.refresh();

   115→  async function handleExport(menuId: string, menuName: string, format: string) {
   116→    try {
   117→      const res = await fetch(`/api/menus/${menuId}/export?format=${format}`);
   118→      if (!res.ok) throw new Error('Error exportando');
   119→      const blob = await res.blob();
   120→      const ext = format === 'excel' ? 'xls' : format === 'word' ? 'doc' : format;
   121→      const url = URL.createObjectURL(blob);
   122→      const a = document.createElement('a');
   123→      a.href = url;
   124→      const safeName = menuName.replace(/[^a-zA-Z0-9áéíóúñÁÉÍÓÚÑ ]/g, '').replace(/\s+/g, '-');
   125→      a.download = `${safeName}.${ext}`;
   126→      document.body.appendChild(a);
   127→      a.click();
   128→      document.body.removeChild(a);
   129→      URL.revokeObjectURL(url);
   130→      toast.success(`Exportado como .${ext}`);
   131→    } catch {
   132→      toast.error('Error al exportar');
   133→    }
   134→  }

   136→  async function handleImport(menuId: string) {

   141→  async function processImport(e: React.ChangeEvent<HTMLInputElement>) {
   142→    const file = e.target.files?.[0];
   143→    if (!file || !importTarget) return;
   144→
   145→    setImporting(true);
   146→    try {
   147→      const text = await file.text();
   148→      const fileType = detectFileType(file.name, text);
   149→
   150→      if (fileType === 'unknown') {
   151→        toast.error('Formato no reconocido. Usa JSON, CSV o Excel.');
   152→        setImporting(false);
   153→        return;
   154→      }
   155→
   156→      const formData = new FormData();
   157→      formData.append('file', file);
   158→
   159→      const res = await fetch(`/api/menus/${importTarget}/import`, {
   160→        method: 'POST',
   161→        body: formData,
   162→      });
   163→
   164→      const data = await res.json();
   165→      if (!res.ok) throw new Error(data.error || 'Error importando');
   166→
   167→      toast.success(data.message || `Importados: ${data.imported?.categories} categorías, ${data.imported?.dishes} platos`);
   168→      router.refresh();
   169→    } catch (err) {
   170→      toast.error(err instanceof Error ? err.message : 'Error al importar');
   171→    } finally {
   172→      setImporting(false);
   173→      setImportTarget(null);
   174→      if (importFileRef.current) importFileRef.current.value = '';
   175→    }
   176→  }
