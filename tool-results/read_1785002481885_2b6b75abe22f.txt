     3→/**
     4→ * Construye el HTML completo autocontenido del menú público.
     5→ * Usa concatenación de strings (no template literals anidados) para evitar
     6→ * problemas de escape. Inyecta los datos via JSON.stringify().
     7→ *
     8→ * Soporta tema personalizables (plan Pro): layout, image_size, card_style,
     9→ * font, secondary color, dark mode, cover image, search, etc.
    10→ *
    11→ * NOVEDADES:
    12→ *  - Cover image como fondo detrás del header de perfil (efecto ultra pro)
    13→ *  - Redes sociales con iconos SVG premium (Facebook, Instagram, WhatsApp,
    14→ *    TikTok, Twitter/X, YouTube, Web)
    15→ *  - Lightbox carrusel: al hacer clic en un plato se abre un overlay con
    16→ *    imagen grande + info detallada. Si el plato no tiene imagen, se muestra
    17→ *    un placeholder elegante.
    18→ *  - Fix del bug "sans is not defined" (property CSS sin comillas)
    19→ */
    20→export function buildMenuHTML(data: MenuData): string {
    21→  // Defaults seguros (plan Free = single col, medium img, expanded card, dark)
    22→  const layout = data.theme_layout || 'single';
    23→  const imageSize = data.theme_image_size || 'medium';
    24→  const cardStyle = data.theme_card_style || 'expanded';
    25→  const font = data.theme_font || 'Inter';
    26→  const darkMode = data.theme_dark_mode !== false; // default true
    27→  const showSearch = data.theme_show_search !== false;
    28→  const showCatIcons = data.theme_show_category_icons !== false;
    29→  const rounded = data.theme_rounded_corners !== false;
    30→  const coverUrl = data.theme_cover_url || null;
    31→  const secondary = data.theme_color_secondary || '#1a1a2e';
    32→  const showGallery = data.theme_dish_gallery !== false; // default true
    33→
    34→  const css = buildCSS({ layout, imageSize, cardStyle, font, darkMode, showSearch, showCatIcons, rounded, coverUrl, secondary, showGallery });
    35→  const js = buildJS({ layout, imageSize, cardStyle, showSearch, showGallery });
    36→  const colorRgb = hexToRgbStr(data.color);
    37→  const secondaryRgb = hexToRgbStr(secondary);
    38→
    39→  // Font CDN (solo si no es Inter/system)
    40→  const fontImport = ['Inter', 'system-ui', '-apple-system'].includes(font)
    41→    ? ''
    42→    : `<link rel="preconnect" href="https://fonts.googleapis.com"><link rel="preconnect" href="https://fonts.gstatic.com" crossorigin><link href="https://fonts.googleapis.com/css2?family=${encodeURIComponent(font)}:wght@400;500;600;700;800&display=swap" rel="stylesheet">`;
    43→
    44→  // ⚠️ FIX CRITICAL: el valor de la property CSS debe ser un JS string completo
    45→  // y CORRECTAMENTE ESCAPADO. Antes generábamos "Inter, "Inter", sans-serif"
    46→  // lo que rompía el parser JS y mataba todo el script (ReferenceError: sans).
    47→  // Usamos JSON.stringify para escapar comillas internas correctamente.
    48→  const fontMainJs = JSON.stringify(font + ', "Inter", sans-serif');
    49→
    50→  let html = '';
    51→  html += '<!DOCTYPE html>\n';
    52→  html += '<html lang="es">\n';
    53→  html += '<head>\n';
    54→  html += '<meta charset="UTF-8">\n';
    55→  html += '<meta name="viewport" content="width=device-width, initial-scale=1.0, viewport-fit=cover">\n';
    56→  html += '<meta name="theme-color" content="' + data.color + '">\n';
    57→  html += '<meta name="description" content="' + escapeHtml(data.description || data.name) + '">\n';
    58→  html += '<title>' + escapeHtml(data.name) + '</title>\n';
    59→  html += fontImport + '\n';

    63→  html += '</head>\n';
    64→  html += '<body>\n';
    65→  html += '<div id="app"></div>\n';
    66→  // Mobile bottom nav (Inicio/Buscar/Favoritos/Carrito) — solo visible en mobile via CSS
    67→  html += '<nav class="mobile-bottom-nav" id="mobileBottomNav" aria-label="Navegación móvil">\n';
    68→  html += '  <button class="mbn-item active" data-action="home" aria-label="Inicio">\n';
    69→  html += '    <span class="mbn-icon"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M3 9l9-7 9 7v11a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2z"/><polyline points="9 22 9 12 15 12 15 22"/></svg></span>\n';
    70→  html += '    <span>Inicio</span>\n';
    71→  html += '  </button>\n';
    72→  html += '  <button class="mbn-item" data-action="search" aria-label="Buscar">\n';
    73→  html += '    <span class="mbn-icon"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><circle cx="11" cy="11" r="8"/><path d="m21 21-4.3-4.3"/></svg></span>\n';
    74→  html += '    <span>Buscar</span>\n';
    75→  html += '  </button>\n';
    76→  html += '  <button class="mbn-item" data-action="favorites" aria-label="Favoritos">\n';
    77→  html += '    <span class="mbn-icon"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M20.84 4.61a5.5 5.5 0 0 0-7.78 0L12 5.67l-1.06-1.06a5.5 5.5 0 0 0-7.78 7.78l1.06 1.06L12 21.23l7.78-7.78 1.06-1.06a5.5 5.5 0 0 0 0-7.78z"/></svg></span>\n';
    78→  html += '    <span class="mbn-badge" id="mbnFavCount" style="display:none;">0</span>\n';

    80→  html += '  </button>\n';

    82→  html += '    <span class="mbn-cart-total" id="mbnCartTotal"></span>\n';
    83→  html += '    <span class="mbn-icon"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><circle cx="9" cy="21" r="1"/><circle cx="20" cy="21" r="1"/><path d="M1 1h4l2.68 13.39a2 2 0 0 0 2 1.61h9.72a2 2 0 0 0 2-1.61L23 6H6"/></svg></span>\n';
    84→  html += '    <span class="mbn-badge" id="mbnCartCount" style="display:none;">0</span>\n';
    85→  html += '    <span>Pedido</span>\n';
    86→  html += '  </button>\n';
    87→  html += '</nav>\n';
    88→  html += '<script>\n';
    89→  html += 'var RESTAURANT = ' + JSON.stringify(data) + ';\n';
    90→  html += 'var SHOW_BRANDING = ' + (!!data.branding_text) + ';\n';
    91→  html += 'var BRANDING_TEXT = ' + JSON.stringify(data.branding_text || '') + ';\n';
    92→  html += 'var THEME = ' + JSON.stringify({ layout, imageSize, cardStyle, showSearch, showGallery }) + ';\n';
    93→  html += 'document.documentElement.style.setProperty("--accent", "' + data.color + '");\n';
    94→  html += 'document.documentElement.style.setProperty("--accent-rgb", "' + colorRgb + '");\n';
    95→  html += 'document.documentElement.style.setProperty("--secondary", "' + secondary + '");\n';
    96→  html += 'document.documentElement.style.setProperty("--secondary-rgb", "' + secondaryRgb + '");\n';
    97→  // ⚠️ FIX: property CSS value debe ser un string COMPLETO entre comillas
    98→  html += 'document.documentElement.style.setProperty("--font-main", ' + fontMainJs + ');\n';
    99→  html += js + '\n';
   100→  html += '</scr' + 'ipt>\n';
   101→  // Swiper JS bundle (carga async para no bloquear el render del menú)
   102→  html += '<script src="https://cdn.jsdelivr.net/npm/swiper@11/swiper-bundle.min.js" referrerpolicy="no-referrer" defer></scr' + 'ipt>\n';
   103→  html += '</body>\n';
   104→  html += '</html>';
   105→  return html;
   106→}

   108→interface ThemeOpts {
   109→  layout: 'single' | 'double' | 'grid';
   110→  imageSize: 'none' | 'small' | 'medium' | 'large' | 'hero';
   111→  cardStyle: 'compact' | 'expanded' | 'minimal';
   112→  font: string;
   113→  darkMode: boolean;
   114→  showSearch: boolean;
   115→  showCatIcons: boolean;
   116→  rounded: boolean;
   117→  coverUrl: string | null;
   118→  secondary: string;
   119→  showGallery: boolean;
   120→}

   122→function buildCSS(opts: ThemeOpts): string {
   123→  const { layout, imageSize, cardStyle, font, darkMode, showSearch, showCatIcons, rounded, coverUrl, showGallery } = opts;
   124→  const radius = rounded ? '16px' : '4px';
   125→  const radiusSm = rounded ? '12px' : '2px';
   126→  const radiusLg = rounded ? '24px' : '8px';
   127→
   128→  // Colores según dark/light — light mode usa beige/cream suave (no blanco puro)
   129→  // para dar calidez tipo PedidosYa/Rappi light theme.
   130→  // --accent-text es la versión oscurecida del accent para texto sobre fondo claro
   131→  // (garantiza contraste incluso si el usuario elige un accent muy claro como amarillo).
   132→  const bg0 = darkMode ? '#07070b' : '#fefcf7';
   133→  const bg1 = darkMode ? '#0f0f1a' : '#ffffff';
   134→  const text = darkMode ? '#f4f4fa' : '#1a1a2e';
   135→  const textMuted = darkMode ? '#8a8a9a' : '#6a6a7a';
   136→  const textSoft = darkMode ? '#b8b8c8' : '#3a3a4a';
   137→  const glass = darkMode ? 'rgba(255,255,255,0.035)' : 'rgba(0,0,0,0.025)';
   138→  const glassStrong = darkMode ? 'rgba(255,255,255,0.07)' : 'rgba(0,0,0,0.05)';
   139→  const border = darkMode ? 'rgba(255,255,255,0.08)' : 'rgba(0,0,0,0.08)';
   140→  const borderStrong = darkMode ? 'rgba(255,255,255,0.14)' : 'rgba(0,0,0,0.14)';
   141→
   142→  let c = '';
   143→  // En light mode, --accent-text es el accent oscurecido 18% (legible sobre beige/cream).
   144→  // En dark mode, --accent-text = --accent tal cual (suficiente contraste sobre oscuro).

'var(--accent)' : 'color-mix(in srgb, var(--accent) 78%, #000)'};}`;
   146→  c += '*{margin:0;padding:0;box-sizing:border-box;-webkit-tap-highlight-color:transparent;}';
