     1→import { escapeHtml, hexToRgbStr, type MenuData } from '@/lib/menu-utils';
     2→
     3→/**
     4→ * Construye el HTML completo autocontenido del menú público.
     5→ * Usa concatenación de strings (no template literals anidados) para evitar
     6→ * problemas de escape. Inyecta los datos via JSON.stringify().
     7→ *
     8→ * Soporta tema personalizables (plan Pro): layout, image_size, card_style,
     9→ * font, secondary color, dark mode, cover image, search, etc.
    10→ *
    11→ * NOVEDADES:
    12→ *  - Cover image como fondo detrás del header de perfil (efecto ultra pro)
    13→ *  - Redes sociales con iconos SVG premium (Facebook, Instagram, WhatsApp,
    14→ *    TikTok, Twitter/X, YouTube, Web)
    15→ *  - Lightbox carrusel: al hacer clic en un plato se abre un overlay con
    16→ *    imagen grande + info detallada. Si el plato no tiene imagen, se muestra
    17→ *    un placeholder elegante.
    18→ *  - Fix del bug "sans is not defined" (property CSS sin comillas)
    19→ */
    20→export function buildMenuHTML(data: MenuData): string {
    21→  // Defaults seguros (plan Free = single col, medium img, expanded card, dark)
    22→  const layout = data.theme_layout || 'single';
    23→  const imageSize = data.theme_image_size || 'medium';
    24→  const cardStyle = data.theme_card_style || 'expanded';
    25→  const font = data.theme_font || 'Inter';
    26→  const darkMode = data.theme_dark_mode !== false; // default true
    27→  const showSearch = data.theme_show_search !== false;
    28→  const showCatIcons = data.theme_show_category_icons !== false;
    29→  const rounded = data.theme_rounded_corners !== false;
    30→  const coverUrl = data.theme_cover_url || null;
    31→  const secondary = data.theme_color_secondary || '#1a1a2e';
    32→  const showGallery = data.theme_dish_gallery !== false; // default true
    33→
    34→  const css = buildCSS({ layout, imageSize, cardStyle, font, darkMode, showSearch, showCatIcons, rounded, coverUrl, secondary, showGallery });
    35→  const js = buildJS({ layout, imageSize, cardStyle, showSearch, showGallery });
    36→  const colorRgb = hexToRgbStr(data.color);
    37→  const secondaryRgb = hexToRgbStr(secondary);
    38→
    39→  // Font CDN (solo si no es Inter/system)
    40→  const fontImport = ['Inter', 'system-ui', '-apple-system'].includes(font)
    41→    ? ''
    42→    : `<link rel="preconnect" href="https://fonts.googleapis.com"><link rel="preconnect" href="https://fonts.gstatic.com" crossorigin><link href="https://fonts.googleapis.com/css2?family=${encodeURIComponent(font)}:wght@400;500;600;700;800&display=swap" rel="stylesheet">`;
    43→
    44→  // ⚠️ FIX CRITICAL: el valor de la property CSS debe ser un JS string completo
    45→  // y CORRECTAMENTE ESCAPADO. Antes generábamos "Inter, "Inter", sans-serif"
    46→  // lo que rompía el parser JS y mataba todo el script (ReferenceError: sans).
    47→  // Usamos JSON.stringify para escapar comillas internas correctamente.
    48→  const fontMainJs = JSON.stringify(font + ', "Inter", sans-serif');
    49→
    50→  let html = '';
    51→  html += '<!DOCTYPE html>\n';
    52→  html += '<html lang="es">\n';
    53→  html += '<head>\n';
    54→  html += '<meta charset="UTF-8">\n';
    55→  html += '<meta name="viewport" content="width=device-width, initial-scale=1.0, viewport-fit=cover">\n';
    56→  html += '<meta name="theme-color" content="' + data.color + '">\n';
    57→  html += '<meta name="description" content="' + escapeHtml(data.description || data.name) + '">\n';
    58→  html += '<title>' + escapeHtml(data.name) + '</title>\n';
    59→  html += fontImport + '\n';
    60→  html += '<style>' + css + '</style>\n';
    61→  html += '</head>\n';
    62→  html += '<body>\n';
    63→  html += '<div id="app"></div>\n';
    64→  html += '<script>\n';
    65→  html += 'var RESTAURANT = ' + JSON.stringify(data) + ';\n';
    66→  html += 'var SHOW_BRANDING = ' + (!!data.branding_text) + ';\n';
    67→  html += 'var BRANDING_TEXT = ' + JSON.stringify(data.branding_text || '') + ';\n';
    68→  html += 'var THEME = ' + JSON.stringify({ layout, imageSize, cardStyle, showSearch, showGallery }) + ';\n';
    69→  html += 'document.documentElement.style.setProperty("--accent", "' + data.color + '");\n';
    70→  html += 'document.documentElement.style.setProperty("--accent-rgb", "' + colorRgb + '");\n';
    71→  html += 'document.documentElement.style.setProperty("--secondary", "' + secondary + '");\n';
    72→  html += 'document.documentElement.style.setProperty("--secondary-rgb", "' + secondaryRgb + '");\n';
    73→  // ⚠️ FIX: property CSS value debe ser un string COMPLETO entre comillas
    74→  html += 'document.documentElement.style.setProperty("--font-main", ' + fontMainJs + ');\n';
    75→  html += js + '\n';
    76→  html += '</scr' + 'ipt>\n';
    77→  html += '</body>\n';
    78→  html += '</html>';
    79→  return html;
    80→}
