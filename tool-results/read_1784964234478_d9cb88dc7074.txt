   231→   231→  async function handlePublish() {
   232→   232→    setPublishing(true);
   233→   233→    try {
   234→   234→      // Validaciones
   235→   235→      if (!menu.name.trim()) {
   236→   236→        toast.error('Ingresa el nombre del restaurante');
   237→   237→        return;
   238→   238→      }
   239→   239→      if (!menu.whatsapp.trim() || !/^\d{8,15}$/.test(menu.whatsapp)) {
   240→   240→        toast.error('WhatsApp inválido. 8-15 dígitos sin + ni espacios');
   241→   241→        return;
   242→   242→      }
   243→   243→      const validCats = categories.filter((c) => c.name.trim());
   244→   244→      if (validCats.length === 0) {
   245→   245→        toast.error('Agrega al menos una categoría con nombre');
   246→   246→        return;
   247→   247→      }
   248→   248→      for (const c of validCats) {
   249→   249→        const validDishes = c.dishes.filter((d) => d.name.trim());
   250→   250→        if (validDishes.length === 0) {
   251→   251→          toast.error(`La categoría "${c.name}" necesita al menos un plato`);
   252→   252→          return;
   253→   253→        }
   254→   254→        for (const d of validDishes) {
   255→   255→          if (!d.price || isNaN(Number(d.price)) || Number(d.price) < 0) {
   256→   256→            toast.error(`Precio inválido en: ${d.name}`);
   257→   257→            return;
   258→   258→          }
   259→   259→        }
   260→   260→      }
   261→   261→
   262→   262→      // Guardar primero
   263→   263→      const res = await fetch(`/api/menus/${initialMenu.id}`, {
   264→   264→        method: 'PUT',
   265→   265→        headers: { 'Content-Type': 'application/json' },
   266→   266→        body: JSON.stringify({
   267→   267→          ...menu,
   268→   268→          is_published: true,
   269→   269→          categories: categories.map((c) => ({
   270→   270→            name: c.name,
   271→   271→            dishes: c.dishes.map((d) => ({
   272→   272→              name: d.name,
   273→   273→              description: d.description,
   274→   274→              price: d.price,
   275→   275→              image_url: d.image,
   276→   276→            })),
   277→   277→          })),
   278→   278→        }),
   279→   279→      });
   280→   280→      if (!res.ok) {
   281→   281→        const data = await res.json();
   282→   282→        throw new Error(data.error || 'Error');
   283→   283→      }
   284→   284→      setMenu((m) => ({ ...m, is_published: true }));
   285→   285→      dirtyRef.current = false;
   286→   286→      toast.success('¡Menú publicado! Ya es visible públicamente.');
   287→   287→      window.open(`/r/${initialMenu.slug}`, '_blank');
   288→   288→    } catch (err) {
   289→   289→      toast.error(err instanceof Error ? err.message : 'Error');
   290→   290→    } finally {
   291→   291→      setPublishing(false);
   292→   292→    }
   293→   293→  }

   355→   355→  async function handleImport(e: React.ChangeEvent<HTMLInputElement>) {
   356→   356→    const file = e.target.files?.[0];

   365→   365→      });

   368→   368→      toast.success(data.message || `Importados: ${data.imported?.categories} categorías, ${data.imported?.dishes} platos`);
   369→   369→      router.refresh();
   370→   370→      // Reload menu data from server
   371→   371→      const menuRes = await fetch(`/api/menus/${initialMenu.id}`);
   372→   372→      const menuData = await menuRes.json();
   373→   373→      if (menuData.menu) {
   374→   374→        const m = menuData.menu;
   375→   375→        setMenu({

   377→   377→          slogan: m.slogan || '',

   381→   381→          color: m.color || '#ff6b35',
   382→   382→          currency: m.currency || 'S/',
   383→   383→          is_published: m.is_published || false,
   384→   384→        });
   385→   385→        setCategories(
   386→   386→          (m.categories || []).map((c: { id: string; name: string; dishes?: Array<{ id: string; name: string; description: string; price: number; image_url: string }> }) => ({
   387→   387→            id: c.id,

   389→   389→            dishes: (c.dishes || []).map((d) => ({
   390→   390→              id: d.id,

   400→   400→      toast.error(err instanceof Error ? err.message : 'Error al importar');
   401→   401→    } finally {

   403→   403→      if (importFileRef.current) importFileRef.current.value = '';
   404→   404→    }

   407→   407→  async function handleExportMenu(format: string) {
   408→   408→    try {
   409→   409→      const res = await fetch(`/api/menus/${initialMenu.id}/export?format=${format}`);
   410→   410→      if (!res.ok) throw new Error('Error exportando');
   411→   411→      const blob = await res.blob();
   412→   412→      const ext = format === 'excel' ? 'xls' : format === 'word' ? 'doc' : format;
   413→   413→      const url = URL.createObjectURL(blob);
   414→   414→      const a = document.createElement('a');
   415→   415→      a.href = url;
   416→   416→      const safeName = (menu.name || 'menu').replace(/[^a-zA-Z0-9áéíóúñÁÉÍÓÚÑ ]/g, '').replace(/\s+/g, '-');
   417→   417→      a.download = `${safeName}.${ext}`;
   418→   418→      document.body.appendChild(a);
   419→   419→      a.click();
   420→   420→      document.body.removeChild(a);
   421→   421→      URL.revokeObjectURL(url);
   422→   422→      toast.success(`Exportado como .${ext}`);
   423→   423→    } catch {
   424→   424→      toast.error('Error al exportar');
   425→   425→    }
   426→   426→  }
