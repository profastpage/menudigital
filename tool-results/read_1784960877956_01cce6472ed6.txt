   103→type Tab = 'stats' | 'users' | 'domains';
   104→
   105→export function SuperAdminClient({ admin }: { admin: AdminInfo }) {
   106→  const router = useRouter();
   107→  const [activeTab, setActiveTab] = useState<Tab>('stats');
   108→  const [stats, setStats] = useState<Stats | null>(null);
   109→  const [users, setUsers] = useState<UserRow[]>([]);
   110→  const [domains, setDomains] = useState<any[]>([]);
   111→  const [loading, setLoading] = useState(true);
   112→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   113→  const [search, setSearch] = useState('');
   114→  const [page, setPage] = useState(1);
   115→  const [totalPages, setTotalPages] = useState(1);
   116→  const [expandedUser, setExpandedUser] = useState<string | null>(null);
   117→  const [userDetail, setUserDetail] = useState<UserDetail | null>(null);
   118→  const [detailLoading, setDetailLoading] = useState(false);
   119→  const [detailModalOpen, setDetailModalOpen] = useState(false);
   120→
   121→  const fetchData = useCallback(async () => {
   122→    setLoading(true);
   123→    try {
   124→      const params = new URLSearchParams({ tab: activeTab });
   125→      if (activeTab === 'users') {
   126→        params.set('search', search);
   127→        params.set('page', String(page));
   128→      }
   129→      const res = await fetch(`/api/admin?${params}`);
   130→      const data = await res.json();
   131→      if (!res.ok) throw new Error(data.error || 'Error');
   132→
   133→      if (activeTab === 'stats') {
   134→        setStats(data.stats);
   135→      } else if (activeTab === 'users') {
   136→        setUsers(data.users || []);
   137→        setTotalPages(Math.ceil((data.total || 0) / 20));
   138→      } else if (activeTab === 'domains') {
   139→        setDomains(data.domains || []);
   140→      }
   141→    } catch (err) {
   142→      toast.error(err instanceof Error ? err.message : 'Error cargando datos');
   143→    } finally {
   144→      setLoading(false);
   145→    }
   146→  }, [activeTab, search, page]);
   147→
   148→  useEffect(() => {
   149→    fetchData();
   150→  }, [fetchData]);
   151→
   152→  async function handleAction(action: string, userId: string, extra?: Record<string, any>) {
   153→    setActionLoading(`${action}-${userId}`);
   154→    try {
   155→      const res = await fetch('/api/admin', {
   156→        method: 'PUT',
   157→        headers: { 'Content-Type': 'application/json' },
   158→        body: JSON.stringify({ action, userId, ...extra }),
   159→      });
   160→      const data = await res.json();
   161→      if (!res.ok) throw new Error(data.error || 'Error');
   162→
   163→      if (action === 'toggle_plan') {
   164→        toast.success(`Plan cambiado a ${data.newPlan}`);
   165→      } else if (action === 'toggle_super_admin') {
   166→        toast.success(`Super admin: ${data.is_super_admin ? 'activado' : 'desactivado'}`);
   167→      } else if (action === 'toggle_active') {
   168→        toast.success(data.is_active ? 'Usuario reactivado' : 'Usuario desactivado');
   169→      } else {
   170→        toast.success('Acción completada');
   171→      }
   172→      fetchData();
   173→    } catch (err) {
   174→      toast.error(err instanceof Error ? err.message : 'Error');
   175→    } finally {
   176→      setActionLoading(null);
   177→    }
   178→  }
   179→
   180→  async function handleViewUserDetail(userId: string) {
   181→    setDetailLoading(true);
   182→    setDetailModalOpen(true);
   183→    try {
   184→      const res = await fetch(`/api/admin?tab=user_detail&userId=${userId}`);
   185→      const data = await res.json();
   186→      if (!res.ok) throw new Error(data.error || 'Error');
   187→      setUserDetail(data.detail);
   188→    } catch (err) {
   189→      toast.error(err instanceof Error ? err.message : 'Error');
   190→      setDetailModalOpen(false);
   191→    } finally {
   192→      setDetailLoading(false);
   193→    }
   194→  }
   195→
   196→  async function handleDeleteUser(userId: string, email: string) {
   197→    if (!confirm(`¿Eliminar usuario ${email} y TODOS sus menús, categorías, platos, dominios y vistas? Esta acción es IRREVERSIBLE.`)) return;
   198→    await handleAction('delete_user', userId);
   199→  }
   200→
   201→  async function handleLogout() {
   202→    try {
   203→      await fetch('/api/auth/logout', { method: 'POST' });
   204→      router.push('/login');
   205→    } catch (err) {
   206→      router.push('/login');
   207→    }
   208→  }
   209→
   210→  const statCards = stats
   211→    ? [
   212→        { label: 'Total usuarios', value: stats.total_users, icon: Users, color: '#d4af37', sub: `${stats.active_users} activos · ${stats.banned_users} baneados` },
   213→        { label: 'Usuarios Pro', value: stats.pro_users, icon: Crown, color: '#22c55e', sub: `${stats.free_users} gratis` },
   214→        { label: 'Registros 7 días', value: stats.recent_signups_7d, icon: TrendingUp, color: '#3b82f6', sub: `${stats.recent_signups_30d} en 30 días` },
   215→        { label: 'Total menús', value: stats.total_menus, icon: MenuIcon, color: '#f97316', sub: `${stats.published_menus} publicados` },
   216→        { label: 'Total platos', value: stats.total_dishes, icon: Hash, color: '#a855f7', sub: `${stats.total_categories} categorías` },
   217→        { label: 'Vistas totales', value: stats.total_views, icon: Eye, color: '#ec4899', sub: 'Acumulado global' },
   218→        { label: 'Dominios', value: `${stats.verified_domains}/${stats.total_domains}`, icon: Globe, color: '#14b8a6', sub: 'verificados/total' },
   219→        { label: 'Super admins', value: stats.super_admins, icon: Shield, color: '#ef4444', sub: 'con acceso total' },
   220→      ]
   221→    : [];
   222→
   223→  return (
   224→    <div className="min-h-screen bg-[#050507] text-white">
   225→      {/* TOP NAV — Distinto al dashboard normal */}
   226→      <header className="border-b border-amber-500/20 bg-gradient-to-r from-[#0a0a14] via-[#0f0a1a] to-[#0a0a14] sticky top-0 z-40">
   227→        <div className="max-w-7xl mx-auto px-6 py-3 flex items-center justify-between">
   228→          <div className="flex items-center gap-3">
   229→            <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-amber-500 to-amber-700 flex items-center justify-center shadow-lg shadow-amber-500/30">
   230→              <Shield className="w-5 h-5 text-black" />
   231→            </div>
   232→            <div>
   233→              <div className="font-bold text-base leading-tight flex items-center gap-2">
   234→                MenuPro · Super Admin
   235→                <span className="px-2 py-0.5 rounded-full bg-amber-500/20 text-amber-400 text-[10px] font-bold border border-amber-500/30">
   236→                  MODO DIOS
   237→                </span>
   238→              </div>
   239→              <div className="text-xs text-white/40 leading-tight">Panel interno · Solo administración</div>
   240→            </div>
   241→          </div>
   242→
   243→          <div className="flex items-center gap-3">
   244→            <div className="flex items-center gap-2 px-3 py-1.5 rounded-lg bg-white/5 border border-white/10">
   245→              {admin.avatar_url ? (
   246→                <img src={admin.avatar_url} alt="" className="w-6 h-6 rounded-full" />
   247→              ) : (
   248→                <div className="w-6 h-6 rounded-full bg-gradient-to-br from-amber-500 to-amber-700 flex items-center justify-center text-[10px] font-bold text-black">
   249→                  {admin.name[0]?.toUpperCase()}
   250→                </div>
   251→              )}
   252→              <span className="text-xs text-white/70 hidden md:inline">{admin.email}</span>
   253→            </div>
   254→            <a
   255→              href="/dashboard"
   256→              className="text-xs text-white/40 hover:text-white px-3 py-1.5 rounded-lg border border-white/10 hover:bg-white/5 transition"
   257→              title="Ir al dashboard de usuario normal"
   258→            >
   259→              Ver dashboard
   260→            </a>
   261→            <Button
   262→              variant="ghost"
   263→              size="sm"
   264→              onClick={handleLogout}
   265→              className="text-red-400/70 hover:text-red-400 hover:bg-red-500/10"
   266→            >
   267→              <LogOut className="w-4 h-4" />
   268→              <span className="hidden md:inline">Salir</span>
   269→            </Button>
   270→          </div>
   271→        </div>
   272→      </header>
   273→
   274→      {/* TABS */}
   275→      <div className="border-b border-white/10 bg-[#070710]/80 backdrop-blur">
   276→        <div className="max-w-7xl mx-auto px-6 flex gap-1">
   277→          {([
   278→            { key: 'stats' as Tab, label: 'Estadísticas', icon: LayoutDashboard },
   279→            { key: 'users' as Tab, label: 'Usuarios', icon: Users },
   280→            { key: 'domains' as Tab, label: 'Dominios', icon: Globe },
   281→          ]).map((tab) => (
   282→            <button
   283→              key={tab.key}
   284→              onClick={() => { setActiveTab(tab.key); setPage(1); }}
   285→              className={`flex items-center gap-2 px-5 py-3 text-sm font-medium border-b-2 transition-colors ${
   286→                activeTab === tab.key
   287→                  ? 'border-amber-500 text-amber-400'
   288→                  : 'border-transparent text-white/50 hover:text-white/80'
   289→              }`}
   290→            >
   291→              <tab.icon className="w-4 h-4" />
   292→              {tab.label}
   293→            </button>
   294→          ))}
   295→        </div>
   296→      </div>
   297→
   298→      <main className="max-w-7xl mx-auto px-6 py-8">
   299→        {/* STATS TAB */}
   300→        {activeTab === 'stats' && (
   301→          loading && !stats ? (
   302→            <div className="flex items-center justify-center py-20">
   303→              <Loader2 className="w-8 h-8 animate-spin text-amber-400" />
   304→            </div>
   305→          ) : stats ? (
   306→            <>
   307→              {/* Top stat cards */}
   308→              <div className="grid grid-cols-2 md:grid-cols-4 gap-4 mb-8">
   309→                {statCards.map((card) => (
   310→                  <div
   311→                    key={card.label}
   312→                    className="bg-gradient-to-br from-white/[0.04] to-transparent border border-white/10 rounded-2xl p-5 hover:border-white/20 transition-colors"
   313→                  >
   314→                    <div className="flex items-center gap-2 mb-2">
   315→                      <card.icon className="w-4 h-4" style={{ color: card.color }} />
   316→                      <span className="text-xs text-white/50">{card.label}</span>
   317→                    </div>
   318→                    <div className="text-2xl font-bold">{card.value}</div>
   319→                    <div className="text-[10px] text-white/30 mt-1">{card.sub}</div>
   320→                  </div>
   321→                ))}
   322→              </div>
   323→
   324→              {/* Revenue block */}
   325→              <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mb-8">
   326→                <div className="bg-gradient-to-br from-amber-500/10 to-transparent border border-amber-500/30 rounded-2xl p-6">
   327→                  <h3 className="font-semibold mb-3 flex items-center gap-2">
   328→                    <DollarSign className="w-5 h-5 text-amber-400" />
   329→                    Ingresos estimados
   330→                  </h3>
   331→                  <div className="text-4xl font-bold text-amber-400">
   332→                    S/ {stats.revenue_estimate_pen}
   333→                    <span className="text-base text-white/40 ml-2">
   334→                      ≈ ${stats.revenue_estimate_usd} USD
   335→                    </span>
   336→                  </div>
   337→                  <p className="text-xs text-white/50 mt-2">
   338→                    {stats.pro_users} suscriptores Pro × S/35/mes
   339→                  </p>
   340→                </div>
   341→
   342→                <div className="bg-gradient-to-br from-emerald-500/10 to-transparent border border-emerald-500/30 rounded-2xl p-6">
   343→                  <h3 className="font-semibold mb-3 flex items-center gap-2">
   344→                    <Activity className="w-5 h-5 text-emerald-400" />
   345→                    Actividad reciente
   346→                  </h3>
   347→                  <div className="space-y-2">
   348→                    <div className="flex justify-between">
   349→                      <span className="text-white/60 text-sm">Registros esta semana</span>
   350→                      <span className="font-bold text-emerald-400">{stats.recent_signups_7d}</span>
   351→                    </div>
   352→                    <div className="flex justify-between">
   353→                      <span className="text-white/60 text-sm">Registros este mes</span>
   354→                      <span className="font-bold text-emerald-400">{stats.recent_signups_30d}</span>
   355→                    </div>
   356→                    <div className="flex justify-between">
   357→                      <span className="text-white/60 text-sm">Vistas totales</span>
   358→                      <span className="font-bold text-emerald-400">{stats.total_views.toLocaleString()}</span>
   359→                    </div>
   360→                  </div>
   361→                </div>
   362→              </div>
   363→
   364→              {/* Top menus by views */}
   365→              <div className="bg-white/[0.03] border border-white/10 rounded-2xl p-6">
   366→                <h3 className="font-semibold mb-4 flex items-center gap-2">
   367→                  <TrendingUp className="w-5 h-5 text-amber-400" />
   368→                  Top 10 menús más vistos
   369→                </h3>
   370→                {stats.top_menus_by_views && stats.top_menus_by_views.length > 0 ? (
   371→                  <div className="space-y-2">
   372→                    {stats.top_menus_by_views.map((m, idx) => (
   373→                      <div key={m.id} className="flex items-center justify-between p-3 bg-white/[0.02] rounded-lg hover:bg-white/[0.04] transition-colors">
   374→                        <div className="flex items-center gap-3">
   375→                          <span className={`w-7 h-7 rounded-full flex items-center justify-center text-xs font-bold ${
   376→                            idx === 0 ? 'bg-amber-500/30 text-amber-300' :
   377→                            idx === 1 ? 'bg-gray-400/20 text-gray-300' :
   378→                            idx === 2 ? 'bg-orange-700/30 text-orange-300' :
   379→                            'bg-white/5 text-white/40'
   380→                          }`}>
   381→                            {idx + 1}
   382→                          </span>
   383→                          <div>
   384→                            <div className="font-medium text-sm">{m.name}</div>
   385→                            <div className="text-xs text-white/40">{m.owner_email}</div>
   386→                          </div>
   387→                        </div>
   388→                        <div className="flex items-center gap-4">
   389→                          <span className="text-xs text-white/40 hidden md:inline">/{m.slug}</span>
   390→                          <span className="px-2.5 py-1 rounded-full bg-emerald-500/20 text-emerald-400 text-xs font-bold">
   391→                            {m.views_count.toLocaleString()} vistas
   392→                          </span>
   393→                          <a
   394→                            href={`/m/${m.slug}`}
   395→                            target="_blank"
   396→                            rel="noreferrer"
   397→                            className="text-white/40 hover:text-white"
   398→                          >
   399→                            <ExternalLink className="w-4 h-4" />
   400→                          </a>
   401→                        </div>
   402→                      </div>
   403→                    ))}
   404→                  </div>
   405→                ) : (
   406→                  <p className="text-sm text-white/40 text-center py-6">No hay datos aún</p>
   407→                )}
   408→              </div>
   409→            </>
   410→          ) : null
   411→        )}
   412→
   413→        {/* USERS TAB */}
   414→        {activeTab === 'users' && (
   415→          <>
   416→            <div className="flex items-center gap-3 mb-6 flex-wrap">
   417→              <div className="relative flex-1 max-w-md">
   418→                <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-white/40" />
   419→                <Input
   420→                  value={search}
   421→                  onChange={(e) => { setSearch(e.target.value); setPage(1); }}
   422→                  placeholder="Buscar por email o nombre..."
   423→                  className="pl-10 bg-white/5 border-white/10 text-white"
   424→                />
   425→              </div>
   426→              <Button
   427→                variant="ghost"
   428→                size="sm"
   429→                onClick={fetchData}
   430→                disabled={loading}
   431→                className="text-white/60"
   432→              >
   433→                <RefreshCw className={`w-4 h-4 mr-1.5 ${loading ? 'animate-spin' : ''}`} />
   434→                Actualizar
   435→              </Button>
   436→              <span className="text-sm text-white/40 ml-auto">
   437→                {users.length} usuario{users.length !== 1 ? 's' : ''} · página {page}
   438→              </span>
   439→            </div>
   440→
   441→            {loading ? (
   442→              <div className="flex items-center justify-center py-20">
   443→                <Loader2 className="w-6 h-6 animate-spin text-amber-400" />
   444→              </div>
   445→            ) : users.length === 0 ? (
   446→              <div className="text-center py-16">
   447→                <Users className="w-12 h-12 text-white/20 mx-auto mb-4" />
   448→                <h3 className="text-lg font-semibold mb-2">Sin usuarios</h3>
   449→                <p className="text-white/50">No se encontraron usuarios con ese criterio.</p>
   450→              </div>
   451→            ) : (
   452→              <>
   453→                <div className="space-y-2">
   454→                  {users.map((u) => (
   455→                    <div
   456→                      key={u.id}
   457→                      className={`bg-white/[0.03] border rounded-xl overflow-hidden transition-colors ${
   458→                        u.is_active === false
   459→                          ? 'border-red-500/40 bg-red-500/[0.03]'
   460→                          : 'border-white/10'
   461→                      }`}
   462→                    >
   463→                      <div
   464→                        className="flex items-center justify-between p-4 cursor-pointer hover:bg-white/[0.02]"
   465→                        onClick={() => setExpandedUser(expandedUser === u.id ? null : u.id)}
   466→                      >
   467→                        <div className="flex items-center gap-3 min-w-0">
   468→                          <div className={`w-10 h-10 rounded-full flex items-center justify-center text-sm font-bold flex-shrink-0 ${
   469→                            u.is_super_admin
   470→                              ? 'bg-gradient-to-br from-amber-500 to-amber-700 text-black'
   471→                              : u.avatar_url
   472→                              ? ''
   473→                              : 'bg-white/10 text-white/60'
   474→                          }`}>
   475→                            {u.avatar_url ? (
   476→                              <img src={u.avatar_url} alt="" className="w-10 h-10 rounded-full object-cover" />
   477→                            ) : (
   478→                              (u.full_name || u.email)[0].toUpperCase()
   479→                            )}
   480→                          </div>
   481→                          <div className="min-w-0">
   482→                            <div className="font-medium text-sm truncate flex items-center gap-2">
   483→                              {u.full_name || 'Sin nombre'}
   484→                              {u.is_super_admin && (
   485→                                <span className="px-1.5 py-0.5 rounded bg-amber-500/20 text-amber-400 text-[10px] font-bold border border-amber-500/30">
   486→                                  SUPER ADMIN
   487→                                </span>
   488→                              )}
   489→                              {u.is_active === false && (
   490→                                <span className="px-1.5 py-0.5 rounded bg-red-500/20 text-red-400 text-[10px] font-bold border border-red-500/30">
   491→                                  BANEADO
   492→                                </span>
   493→                              )}
   494→                              {u.plan === 'pro' && (
   495→                                <span className="px-1.5 py-0.5 rounded bg-emerald-500/20 text-emerald-400 text-[10px] font-bold">
   496→                                  PRO
   497→                                </span>

   499→                            </div>
