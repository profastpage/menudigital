   231→  async function handlePublish() {
   232→    setPublishing(true);
   233→    try {
   234→      // Validaciones
   235→      if (!menu.name.trim()) {
   236→        toast.error('Ingresa el nombre del restaurante');
   237→        return;
   238→      }
   239→      if (!menu.whatsapp.trim() || !/^\d{8,15}$/.test(menu.whatsapp)) {
   240→        toast.error('WhatsApp inválido. 8-15 dígitos sin + ni espacios');
   241→        return;
   242→      }
   243→      const validCats = categories.filter((c) => c.name.trim());
   244→      if (validCats.length === 0) {
   245→        toast.error('Agrega al menos una categoría con nombre');
   246→        return;
   247→      }
   248→      for (const c of validCats) {
   249→        const validDishes = c.dishes.filter((d) => d.name.trim());
   250→        if (validDishes.length === 0) {
   251→          toast.error(`La categoría "${c.name}" necesita al menos un plato`);
   252→          return;
   253→        }
   254→        for (const d of validDishes) {
   255→          if (!d.price || isNaN(Number(d.price)) || Number(d.price) < 0) {
   256→            toast.error(`Precio inválido en: ${d.name}`);
   257→            return;
   258→          }
   259→        }
   260→      }
   261→
   262→      // Guardar primero
   263→      const res = await fetch(`/api/menus/${initialMenu.id}`, {
   264→        method: 'PUT',
   265→        headers: { 'Content-Type': 'application/json' },
   266→        body: JSON.stringify({
   267→          ...menu,
   268→          is_published: true,
   269→          categories: categories.map((c) => ({
   270→            name: c.name,
   271→            dishes: c.dishes.map((d) => ({
   272→              name: d.name,
   273→              description: d.description,
   274→              price: d.price,
   275→              image_url: d.image,
   276→            })),
   277→          })),
   278→        }),
   279→      });
   280→      if (!res.ok) {
   281→        const data = await res.json();
   282→        throw new Error(data.error || 'Error');
   283→      }
   284→      setMenu((m) => ({ ...m, is_published: true }));
   285→      dirtyRef.current = false;
   286→      toast.success('¡Menú publicado! Ya es visible públicamente.');
   287→      window.open(`/r/${initialMenu.slug}`, '_blank');
   288→    } catch (err) {
   289→      toast.error(err instanceof Error ? err.message : 'Error');
   290→    } finally {
   291→      setPublishing(false);
   292→    }
   293→  }

   355→  async function handleImport(e: React.ChangeEvent<HTMLInputElement>) {
   356→    const file = e.target.files?.[0];
   357→    if (!file) return;
   358→    setImporting(true);
   359→    try {
   360→      const formData = new FormData();
   361→      formData.append('file', file);
   362→      const res = await fetch(`/api/menus/${initialMenu.id}/import`, {
   363→        method: 'POST',
   364→        body: formData,
   365→      });
   366→      const data = await res.json();
   367→      if (!res.ok) throw new Error(data.error || 'Error importando');
   368→      toast.success(data.message || `Importados: ${data.imported?.categories} categorías, ${data.imported?.dishes} platos`);
   369→      router.refresh();
   370→      // Reload menu data from server
   371→      const menuRes = await fetch(`/api/menus/${initialMenu.id}`);
   372→      const menuData = await menuRes.json();
   373→      if (menuData.menu) {
   374→        const m = menuData.menu;
   375→        setMenu({
   376→          name: m.name || '',
   377→          slogan: m.slogan || '',
   378→          description: m.description || '',
   379→          whatsapp: m.whatsapp || '',
   380→          logo: m.logo_url || '',
   381→          color: m.color || '#ff6b35',
   382→          currency: m.currency || 'S/',
   383→          is_published: m.is_published || false,
   384→        });
   385→        setCategories(
   386→          (m.categories || []).map((c: { id: string; name: string; dishes?: Array<{ id: string; name: string; description: string; price: number; image_url: string }> }) => ({
   387→            id: c.id,
   388→            name: c.name,
   389→            dishes: (c.dishes || []).map((d) => ({
   390→              id: d.id,
   391→              name: d.name,
   392→              description: d.description || '',
   393→              price: String(d.price),
   394→              image: d.image_url || '',
   395→            })),
   396→          }))
   397→        );
   398→      }
   399→    } catch (err) {
   400→      toast.error(err instanceof Error ? err.message : 'Error al importar');
   401→    } finally {
   402→      setImporting(false);
   403→      if (importFileRef.current) importFileRef.current.value = '';
   404→    }
   405→  }

   407→  async function handleExportMenu(format: string) {
   408→    try {
   409→      const res = await fetch(`/api/menus/${initialMenu.id}/export?format=${format}`);
   410→      if (!res.ok) throw new Error('Error exportando');
   411→      const blob = await res.blob();
   412→      const ext = format === 'excel' ? 'xls' : format === 'word' ? 'doc' : format;
   413→      const url = URL.createObjectURL(blob);
   414→      const a = document.createElement('a');
   415→      a.href = url;
   416→      const safeName = (menu.name || 'menu').replace(/[^a-zA-Z0-9áéíóúñÁÉÍÓÚÑ ]/g, '').replace(/\s+/g, '-');
   417→      a.download = `${safeName}.${ext}`;
   418→      document.body.appendChild(a);
   419→      a.click();
   420→      document.body.removeChild(a);
   421→      URL.revokeObjectURL(url);
   422→      toast.success(`Exportado como .${ext}`);
   423→    } catch {
   424→      toast.error('Error al exportar');
   425→    }
   426→  }
